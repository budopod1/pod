set i = 0
try {
    forever {
        try {
            set j = 0
            forever {
                if (= $j 10) {
                    throw::error (choose (>= $i 3) errb erra)
                }
                pre j + 1
            }
        } catch erra {
            print "caught" $0
        }
        pre i + 1
    }
} catch errb {
    print "done" $0
}
