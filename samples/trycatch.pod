set i = 0
try {
    forever {
        try {
            set j = 0
            forever {
                if (= $j 10) {
                    throw::error (choose (>= $i 3) errb erra)
                }
                set j = (+ $j 1)
            }
        } catch erra {
            print "caught" $0
        }
        set i = (+ 1 $i)
    }
} catch errb {
    print "done" $0
}
